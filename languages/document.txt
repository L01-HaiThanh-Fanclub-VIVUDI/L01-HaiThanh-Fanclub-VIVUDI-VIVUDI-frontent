    name: 'Tên',
    email: 'Email',
};

/******************************************************************************
 * GHI CHÚ                                                                    *
 * - `t()` hỗ trợ key dạng lồng nhau: `module.key1.key2`                      *
 * - `getMessage()` giúp format với tham số động cho MSG_ID thông thường      *
 * - `getAPIMessage()` dùng cho MSG_ID dạng object chứa nhiều MSG_API_ID      *
 * - Không cần hardcode message ở nhiều nơi, dễ mở rộng và bảo trì            *
 * - Key trong ngôn ngữ nên nhất quán, tránh lỗi runtime                      *
 ******************************************************************************/
